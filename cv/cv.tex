% ------------------------------ building blocks ----------------------------
\newlength{\cvdate}\setlength{\cvdate}{1.27in}   % date column + gutter
\newlength{\listind}\setlength{\listind}{0.3in}  % hanging indent, lists
\newlength{\pubind}\setlength{\pubind}{0.4in}    % hanging indent, publications
\newcommand{\eduind}{0.33in}                      % year indent (Education)

% date column + hanging description; \dentedi indents the year (Education)
\newcommand{\dented}[2]{\par\noindent\hangindent=\cvdate\hangafter=1\relax
  \makebox[\cvdate][l]{#1}#2\par\addvspace{2pt}}
\newcommand{\dentedi}[2]{\par\noindent\hangindent=\cvdate\hangafter=1\relax
  \makebox[\cvdate][l]{\hspace{\eduind}#1}#2\par\addvspace{2pt}}

% hanging list item / publication
\newcommand{\hitem}[1]{\par\noindent\hangindent=\listind\hangafter=1\relax
  #1\par\addvspace{2pt}}
\newcommand{\pub}[1]{\par\noindent\hangindent=\pubind\hangafter=1\relax
  #1\par\addvspace{6pt}}

% section + subheading
\newcommand{\cvsection}[1]{\addvspace{12pt}\par\noindent
  {\bfseries\large #1}\par\nobreak\addvspace{3pt}}
\newcommand{\cvsectionc}[1]{\addvspace{12pt}\par\nobreak
  {\centering\bfseries\large #1\par}\nobreak\addvspace{2pt}}
\newcommand{\cvsubi}[1]{\addvspace{5pt}\par\noindent
  {\bfseries\itshape #1}\par\nobreak\addvspace{1pt}}
\newcommand{\cvsubb}[1]{\addvspace{5pt}\par\noindent
  {\bfseries #1}\par\nobreak\addvspace{1pt}}
\newcommand{\prose}[1]{\par\noindent #1\par}
